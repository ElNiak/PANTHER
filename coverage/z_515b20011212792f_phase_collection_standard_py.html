133 @classmethod 

134 def get_patterns_for_service( 

135 cls, 

136 protocol: str, 

137 service_type: str, 

138 service_name: Optional[str] = None, 

139 language: Optional[str] = None, 

140 ) -> List[Tuple[str, str]]: 

141 """Get complete pattern set for a service.""" 

142 patterns = [] 

143 

144 # Add base patterns 

145 patterns.extend(cls.BASE_PHASE_PATTERNS) 

146 

147 # Add protocol-specific patterns 

148 if protocol.lower() in cls.PROTOCOL_PATTERNS: 

149 patterns.extend(cls.PROTOCOL_PATTERNS[protocol.lower()]) 

150 

151 # Add service type patterns 

152 if service_type.lower() in cls.SERVICE_TYPE_PATTERNS: 

153 patterns.extend(cls.SERVICE_TYPE_PATTERNS[service_type.lower()]) 

154 

155 # Add language-specific patterns 

156 if language and language.lower() in cls.LANGUAGE_PATTERNS: 

157 patterns.extend(cls.LANGUAGE_PATTERNS[language.lower()]) 

158 

159 # Substitute service name placeholders 

160 if service_name: 

161 patterns = [ 

162 (output_type, pattern.replace("{service_name}", service_name)) 

163 for output_type, pattern in patterns 

164 ] 

165 

166 return patterns

168 @classmethod 

169 def validate_patterns( 

170 cls, patterns: List[Tuple[str, str]] 

171 ) -> Tuple[bool, List[str]]: 

172 """Validate patterns follow expected format.""" 

173 errors = [] 

174 

175 if not patterns: 

176 errors.append("No patterns provided") 

177 return False, errors 

178 

179 required_phases = {"pre-compile", "compile", "runtime", "test"} 

180 found_phases = set() 

181 

182 for output_type, pattern in patterns: 

183 if not isinstance(output_type, str) or not isinstance(pattern, str): 

184 errors.append(f"Invalid pattern format: {output_type}, {pattern}") 

185 continue 

186 

187 # Check for phase coverage 

188 for phase in required_phases: 

189 if phase in pattern: 

190 found_phases.add(phase) 

191 

192 missing_phases = required_phases - found_phases 

193 if missing_phases: 

194 errors.append(f"Missing required phases: {missing_phases}") 

195 

196 return len(errors) == 0, errors
