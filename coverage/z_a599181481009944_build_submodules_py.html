13def detect_openssl_prefix() -> str: 

14 """Auto-detect OpenSSL installation prefix. 

15 

16 Detection priority: 

17 1. OPENSSL_PREFIX env var (explicit override) 

18 2. brew --prefix openssl (Intel + Apple Silicon macOS) 

19 3. pkg-config --variable=prefix openssl (Linux) 

20 4. Filesystem fallbacks 

21 """ 

22 # 1. Explicit env var 

23 env_prefix = os.environ.get("OPENSSL_PREFIX") 

24 if env_prefix and os.path.isdir(env_prefix): 

25 print(f"OpenSSL prefix from OPENSSL_PREFIX env: {env_prefix}") 

26 return env_prefix 

27 

28 # 2. Homebrew (macOS) 

29 if shutil.which("brew"): 

30 try: 

31 result = subprocess.run( 

32 ["brew", "--prefix", "openssl"], 

33 capture_output=True, 

34 text=True, 

35 timeout=10, 

36 ) 

37 if result.returncode == 0: 

38 prefix = result.stdout.strip() 

39 if prefix and os.path.isdir(prefix): 

40 print(f"OpenSSL prefix from brew: {prefix}") 

41 return prefix 

42 except (subprocess.TimeoutExpired, OSError): 

43 pass 

44 

45 # 3. pkg-config (Linux) 

46 if shutil.which("pkg-config"): 

47 try: 

48 result = subprocess.run( 

49 ["pkg-config", "--variable=prefix", "openssl"], 

50 capture_output=True, 

51 text=True, 

52 timeout=10, 

53 ) 

54 if result.returncode == 0: 

55 prefix = result.stdout.strip() 

56 if prefix and os.path.isdir(prefix): 

57 print(f"OpenSSL prefix from pkg-config: {prefix}") 

58 return prefix 

59 except (subprocess.TimeoutExpired, OSError): 

60 pass 

61 

62 # 4. Filesystem fallbacks 

63 fallbacks = [ 

64 "/opt/homebrew/opt/openssl@3", # Apple Silicon Homebrew 

65 "/opt/homebrew/opt/openssl", # Apple Silicon Homebrew (unversioned) 

66 "/usr/local/opt/openssl", # Intel Homebrew 

67 "/usr", # System default (Linux) 

68 ] 

69 for path in fallbacks: 

70 if os.path.isdir(path) and ( 

71 os.path.isdir(os.path.join(path, "include")) 

72 or os.path.isdir(os.path.join(path, "lib")) 

73 ): 

74 print(f"OpenSSL prefix from fallback: {path}") 

75 return path 

76 

77 print("WARNING: Could not detect OpenSSL prefix, defaulting to /usr") 

78 return "/usr"

301def install_z3(): 

302 """Install Z3 Python bindings and libraries into ivy/ directory. 

303 

304 For legacy builds (BUILD_MODE=""), make install with --pypkgdir already 

305 placed Python files in ivy/z3/. We still copy .so/.h files. 

306 

307 For CMake builds (non-empty BUILD_MODE), we must also copy the 

308 generated Python binding files (z3core.py, z3consts.py) from the 

309 build directory, plus the source .py files. 

310 """ 

311 import shutil 

312 

313 print("--- Installing Z3 ---") 

314 make_dir_exist("ivy/lib") 

315 make_dir_exist("ivy/z3") 

316 

317 if platform.system() == "Windows": 

318 do_cmd("copy submodules\\z3\\src\\api\\*.h ivy\\include") 

319 do_cmd('copy "submodules\\z3\\src\\api\\c++\\*.h" ivy\\include') 

320 do_cmd("copy submodules\\z3\\build\\*.dll ivy\\lib") 

321 do_cmd("copy submodules\\z3\\build\\*.lib ivy\\lib") 

322 do_cmd("copy submodules\\z3\\build\\*.dll ivy\\z3") 

323 do_cmd("copy submodules\\z3\\build\\python\\z3\\*.py ivy\\z3") 

324 elif platform.system() == "Darwin": 

325 do_cmd("cp include/*.h ivy/include") 

326 do_cmd("cp lib/*.dylib ivy/lib") 

327 do_cmd("cp lib/*.dylib ivy/z3") 

328 else: 

329 do_cmd("cp include/*.h ivy/include") 

330 if list((ROOT / "lib").glob("*.so")): 

331 do_cmd("cp lib/*.so ivy/lib") 

332 do_cmd("cp lib/*.so ivy/z3") 

333 elif list((ROOT / "lib").glob("*.a")): 

334 do_cmd("cp lib/*.a ivy/lib") 

335 

336 # For CMake-based builds (non-empty BUILD_MODE), make install doesn't 

337 # use --pypkgdir, so Python bindings aren't placed in ivy/z3/ automatically. 

338 # Copy them explicitly from the build output and source tree. 

339 if BUILD_MODE in MODES and BUILD_MODE != "": 

340 z3_py_src = SUBMOD / "z3" / "src" / "api" / "python" / "z3" 

341 z3_build_py = SUBMOD / "z3" / "build" / "python" / "z3" 

342 

343 # Copy source Python files (z3.py, __init__.py, z3num.py, etc.) 

344 if z3_py_src.exists(): 

345 print(f"Copying Z3 Python source files from {z3_py_src}") 

346 for py_file in z3_py_src.glob("*.py"): 

347 shutil.copy2(py_file, IVY / "z3" / py_file.name) 

348 

349 # Copy generated files (z3core.py, z3consts.py) from build output 

350 if z3_build_py.exists(): 

351 print(f"Copying Z3 generated Python files from {z3_build_py}") 

352 for py_file in z3_build_py.glob("*.py"): 

353 shutil.copy2(py_file, IVY / "z3" / py_file.name) 

354 else: 

355 print(f"WARNING: Z3 build Python directory not found: {z3_build_py}") 

356 print("z3core.py and z3consts.py may be missing") 

357 

358 # Verify Z3 Python bindings are complete 

359 required_files = ["__init__.py", "z3.py"] 

360 missing = [f for f in required_files if not (IVY / "z3" / f).exists()] 

361 if missing: 

362 print(f"WARNING: Z3 Python binding files missing from ivy/z3/: {missing}") 

363 # Fallback: copy from source tree 

364 z3_py_src = SUBMOD / "z3" / "src" / "api" / "python" / "z3" 

365 if z3_py_src.exists(): 

366 print(f"Fallback: copying from {z3_py_src}") 

367 for py_file in z3_py_src.glob("*.py"): 

368 shutil.copy2(py_file, IVY / "z3" / py_file.name) 

369 # Re-check after fallback — fail hard if bindings are still missing 

370 still_missing = [f for f in required_files if not (IVY / "z3" / f).exists()] 

371 if still_missing: 

372 print( 

373 f"FATAL: Z3 Python bindings still missing after fallback: {still_missing}" 

374 ) 

375 sys.exit(1)
