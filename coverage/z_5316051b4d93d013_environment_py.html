37 def get_plugin_config(self, config_class: Type[T], validate: bool = True) -> T: 

38 """Get typed plugin configuration with defaults. 

39 

40 This method creates an instance of the plugin config class, using values 

41 from plugin_config where available and defaults from the class where not. 

42 

43 For nested Pydantic models, it handles type conversion intelligently. 

44 

45 Args: 

46 config_class: The plugin configuration class 

47 validate: Whether to validate using PluginConfigResolver (if available) 

48 

49 Returns: 

50 Typed plugin configuration instance 

51 """ 

52 # For plugin configs that have complex nested defaults, we need to be careful 

53 # First, create an instance with all defaults 

54 default_instance = config_class() 

55 

56 # Get default values (handle both Pydantic v1 and v2) 

57 try: 

58 default_dict = default_instance.model_dump() 

59 except AttributeError: 

60 default_dict = default_instance.dict() 

61 

62 # Deep merge plugin_config values over defaults with type awareness 

63 merged_config = self._deep_merge_with_type_conversion( 

64 default_dict, self.plugin_config, config_class 

65 ) 

66 

67 # Create final instance with merged values 

68 instance = config_class(**merged_config) 

69 

70 # Optional validation with PluginConfigResolver 

71 if validate and hasattr(self, "type"): 

72 try: 

73 from panther.plugins.core.plugin_config_resolver import ( 

74 get_plugin_config_resolver, 

75 ) 

76 

77 resolver = get_plugin_config_resolver() 

78 

79 # Try to find the expected config class 

80 expected_class = resolver.resolve_environment_config_class( 

81 environment_type=self.type, 

82 category=( 

83 "network" 

84 if isinstance(self, NetworkEnvironmentConfig) 

85 else "execution"

94 logger.warning(

96 f"got {config_class.__name__}" 

97 ) 

98 except Exception as e: 

99 # Validation is optional, so we just log and continue 

100 import logging

103 logger.debug(f"Could not validate plugin config with resolver: {e}")

107 def _deep_merge_configs( 

108 self, defaults: Dict[str, Any], overrides: Dict[str, Any] 

109 ) -> Dict[str, Any]: 

110 """Deep merge configuration dictionaries, preserving structure. 

111 

112 Args: 

113 defaults: Default configuration with full structure 

114 overrides: Override values that may have simplified structure 

115 

116 Returns: 

117 Merged configuration 

118 """ 

119 result = defaults.copy() 

120 

121 for key, value in overrides.items(): 

122 if key in result: 

123 if isinstance(result[key], dict) and isinstance(value, dict): 

124 # Recursive merge for nested dicts 

125 result[key] = self._deep_merge_configs(result[key], value) 

126 else: 

127 # Direct override for non-dict values 

128 result[key] = value 

129 else: 

130 # New key not in defaults 

131 result[key] = value 

132 

133 return result

153 """
