    def check_container_exit_codes(self):
    """
    Checks the exit codes of all containers and reports them in the logs.
    This helps identify failed tests and specific error codes like timeouts (124) or crashes (134).
    Also updates the execution status in the service manager objects.
    """
    self.logger.info("Checking container exit codes")

    try:
        with open(
            os.path.join(self.output_dir, "logs", "container-exit-codes.log"), "w"
        ) as log_file:
            # Get all container IDs for this compose project
            result = subprocess.run(
                [
                    "docker",
                    "compose",
                    "-f",
                    str(self.rendered_services_network_config_file_path),
                    "ps",
                    "-q",
                ],
                capture_output=True,
                text=True,
            )

            if result.returncode != 0:
                self.logger.error(f"Failed to get container IDs: {result.stderr}")
                return

            container_ids = result.stdout.strip().split("\n")
            if not container_ids or (len(container_ids) == 1 and not container_ids[0]):
                self.logger.warning("No containers found to check exit codes")
                return

            # Track overall test success/failure
            has_failures = False
            has_warnings = False  # For non-critical failures
            exit_code_summary = []
            service_status_map = {}

            for container_id in container_ids:
                if not container_id:
                    continue

                # Get container details including exit code
                inspect_result = subprocess.run(
                    [
                        "docker",
                        "inspect",
                        "--format",
                        "{{.Name}},{{.State.ExitCode}},{{.State.Status}}",
                        container_id,
                    ],
                    capture_output=True,
                    text=True,
                )

                if inspect_result.returncode != 0:
                    self.logger.error(f"Failed to inspect container {container_id}: {inspect_result.stderr}")
                    continue

                container_data = inspect_result.stdout.strip().split(",")
                if len(container_data) < 3:
                    continue

                container_name = container_data[0].replace("/", "")
                exit_code = container_data[1]
                status = container_data[2]

                # Map to get the actual service name from the container name
                service_name = container_name.split("-")[-1]  # Extract service name part

                # For running containers, handle status differently
                if status == "running":
                    self.logger.info(f"Container {container_name} is still running")

                    # Check for compilation errors in running containers 
                    # by examining specific log files created during compilation
                    compile_error_log = os.path.join(self.output_dir, "logs", f"{service_name}_compile_error.log")
                    compilation_log = os.path.join(self.output_dir, "logs", f"{service_name}_compilation.log")
                    status_log = os.path.join(self.output_dir, "logs", f"{service_name}_status.log")

                    # Check for explicit compilation failure indicators
                    has_compilation_failure = False

                    if os.path.exists(compile_error_log):
                        self.logger.warning(f"Running container {container_name} has compilation errors logged")
                        has_compilation_failure = True

                    if os.path.exists(status_log):
                        try:
                            with open(status_log, "r") as status_file:
                                status_content = status_file.read()
                                if "ERROR_TYPE=COMPILATION_FAILURE" in status_content:
                                    self.logger.warning(f"Running container {container_name} has compilation failure status")
                                    has_compilation_failure = True
                        except Exception as e:
                            self.logger.error(f"Error reading status log for {container_name}: {str(e)}")

                    # If compilation failures detected, consider the container as failed
                    if has_compilation_failure:
                        self.logger.warning(f"Container {container_name} is marked as failed due to compilation errors")
                        exit_code = "1"  # Mark as failed with exit code 1
                        status = "compilation-failed"  # Custom status
                    else:
                        # Otherwise, just mark as running
                        exit_code = "running"

                try:
                    if exit_code == "running":
                        exit_code_int = None  # Special case for running containers
                    else:
                        exit_code_int = int(exit_code)
                except ValueError:
                    exit_code_int = -1
                    self.logger.warning(f"Container {container_name} has non-integer exit code: {exit_code}")

                # Log container status for debugging
                detailed_status = f"{status}"
                if status == "running":
                    # Check if container is healthy using docker inspect
                    detailed_status = "running (checking for errors in logs)"
                    # Look for errors in the logs if container is still running
                    logs_result = subprocess.run(
                        ["docker", "logs", "--tail", "20", container_id],
                        capture_output=True, text=True
                    )
                    error_keywords = ["error:", "exception:", "fatal:", "failed:"]
                    for keyword in error_keywords:
                        if keyword in logs_result.stdout.lower() or keyword in logs_result.stderr.lower():
                            detailed_status = f"running (potential issues detected: {keyword})"
                            # Save problematic logs for inspection
                            error_log_path = os.path.join(self.output_dir, "logs", f"{service_name}_potential_issues.log")
                            try:
                                with open(error_log_path, "w") as error_file:
                                    error_file.write(logs_result.stdout + "\n\n" + logs_result.stderr)
                            except Exception as e:
                                self.logger.error(f"Failed to write potential issues to log: {str(e)}")
                            break

                self.logger.debug(f"Container {container_name} status: {detailed_status}, exit_code: {exit_code}")

                # Determine status message based on exit code
                status_message = ""
                error_type = None
                if exit_code == "124":
                    status_message = "TIMEOUT"
                    error_type = "TIMEOUT"
                    has_failures = True
                elif exit_code == "134":
                    status_message = "CRASHED (SIGABRT)"
                    error_type = "CRASH_ABORT"
                    has_failures = True
                elif exit_code == "139":
                    status_message = "CRASHED (SIGSEGV)"
                    error_type = "CRASH_SEGV"
                    has_failures = True
                elif exit_code != "0" and exit_code.strip():
                    status_message = f"FAILED (code: {exit_code})"
                    error_type = "GENERAL_FAILURE"
                    has_failures = True
                else:
                    status_message = "SUCCESS"

                # Check if container status files exist and parse them for additional info
                status_file_path = os.path.join(self.output_dir, "logs", f"{service_name}_status.log")
                if os.path.exists(status_file_path):
                    try:
                        with open(status_file_path, "r") as status_file:
                            status_content = status_file.read()
                            # Parse status file for exit code, error type, etc.
                            if "EXIT_CODE=" in status_content:
                                file_exit_code = status_content.split("EXIT_CODE=")[1].split("\n")[0].strip()
                                try:
                                    exit_code_int = int(file_exit_code)
                                    exit_code = file_exit_code  # Use the exit code from the file
                                except ValueError:
                                    pass

                            if "ERROR_TYPE=" in status_content:
                                error_type = status_content.split("ERROR_TYPE=")[1].split("\n")[0].strip()

                            # Extract additional failure details
                            failed_command = None
                            command_type = None
                            command_index = None

                            if "FAILED_COMMAND=" in status_content:
                                # Extract the failed command (might be multi-line)
                                failed_cmd_parts = status_content.split("FAILED_COMMAND=\"")[1].split("\"")
                                if len(failed_cmd_parts) > 1:
                                    failed_command = failed_cmd_parts[0]

                            if "COMMAND_TYPE=" in status_content:
                                command_type = status_content.split("COMMAND_TYPE=")[1].split("\n")[0].strip()

                            if "COMMAND_INDEX=" in status_content:
                                try:
                                    command_index = int(status_content.split("COMMAND_INDEX=")[1].split("\n")[0].strip())
                                except ValueError:
                                    pass

                            # Add detailed error information to service status map if available
                            if failed_command:
                                if service_name not in service_status_map:
                                    service_status_map[service_name] = {}

                                service_status_map[service_name]["failed_command"] = failed_command
                                if command_type:
                                    service_status_map[service_name]["command_type"] = command_type
                                if command_index is not None:
                                    service_status_map[service_name]["command_index"] = command_index
                    except Exception as e:
                        self.logger.error(f"Error reading status file for {service_name}: {str(e)}")

                # Store service status info
                service_status_map[service_name] = {
                    "exit_code": exit_code_int,
                    "error_type": error_type,
                    "status": "failed" if exit_code_int != 0 else "completed",
                    "failed": exit_code_int != 0
                }

                log_message = f"Container {container_name} exited with code {exit_code} ({status_message})"
                exit_code_summary.append(log_message)

                # Log the result
                self.logger.info(log_message)
                log_file.write(f"{log_message}\n")

            # Update service managers with execution status
            for service_manager in self.services_managers:
                service_name = service_manager.get_service_name()
                if service_name in service_status_map:
                    status_info = service_status_map[service_name]

                    # Initialize kwargs for set_execution_status
                    kwargs = {
                        'exit_code': status_info.get("exit_code"),
                        'error_type': status_info.get("error_type"),
                        'failed': status_info.get("failed"),
                        'status': status_info.get("status")
                    }

                    # Add detailed command failure information if available
                    if "failed_command" in status_info:
                        kwargs['failed_command'] = status_info["failed_command"]

                    if "command_type" in status_info:
                        kwargs['command_type'] = status_info["command_type"]

                    if "command_index" in status_info:
                        kwargs['command_index'] = status_info["command_index"]

                    # Update the service manager with all available status information
                    service_manager.set_execution_status(**kwargs)

                    self.logger.info(f"Updated status for service {service_name}: {service_manager.get_execution_status()}")

            # Check for non-critical failures (warnings)
            for service_name in service_status_map.keys():
                if self.has_non_critical_failures(service_name):
                    has_warnings = True
                    self.logger.info(f"Service {service_name} has non-critical warnings")
                    exit_code_summary.append(f"Service {service_name} has non-critical warnings (not affecting test result)")

            # Write summary to a dedicated file
            with open(
                os.path.join(self.output_dir, "logs", "test-exit-codes-summary.log"), "w"
            ) as summary_file:
                summary_file.write("\n".join(exit_code_summary))
                if has_failures:
                    summary_file.write("\n\nTEST EXECUTION FAILED: One or more containers exited with non-zero code\n")
                elif has_warnings:
                    summary_file.write("\n\nTEST EXECUTION PASSED WITH WARNINGS: Non-critical failures detected but test is considered successful\n")
                else:
                    summary_file.write("\n\nTEST EXECUTION SUCCESSFUL: All containers exited normally\n")

            if has_failures:
                self.logger.warning("TEST EXECUTION FAILED: One or more containers exited with non-zero code")
            elif has_warnings:
                self.logger.info("TEST EXECUTION PASSED WITH WARNINGS: Non-critical failures detected but test is considered successful")
            else:
                self.logger.info("TEST EXECUTION SUCCESSFUL: All containers exited normally")

    except Exception as e:
        self.logger.error(f"Error checking container exit codes: {str(e)}\n{traceback.format_exc()}")

    """
    Creates a properly formatted non-critical command from a list of command lines.

    """
    Generates a comprehensive test report summary with service status and error information.
    This consolidates exit codes, error types, and other status information from all services
    into a single summary file for easier debugging.

    Critical command failures mark the test as FAILED, while non-critical command

    """

            summary_file.write("=" * 80 + "\n")
            summary_file.write("TEST EXECUTION SUMMARY REPORT\n")
            summary_file.write("=" * 80 + "\n\n")

            # Overall test status tracking
            overall_critical_failure = False
            overall_non_critical_failure = False

            # Collect service status information
            service_results = []
            for service_manager in self.services_managers:
                service_name = service_manager.get_service_name()
                status = service_manager.get_execution_status()

                # Check for non-critical failures (warning logs)
                has_non_critical_failures = self.has_non_critical_failures(service_name)

                # Track overall failures - critical vs non-critical
                if status.get("failed", False):
                    overall_critical_failure = True
                elif has_non_critical_failures:
                    overall_non_critical_failure = True

                # Format status for reporting - now with WARNING state
                if status.get("failed", False):
                    status_str = "FAILED"  # Critical failure
                elif has_non_critical_failures:
                    status_str = "WARNING"  # Non-critical failures only
                else:
                    status_str = "SUCCESS"  # No failures

                exit_code = status.get("exit_code")
                error_type = status.get("error_type", "NONE")

                # Build detailed service report
                service_report = f"Service: {service_name}\n"
                service_report += f"  Status: {status_str}\n"

                # Handle special case for "None" exit codes - typically running containers 
                if exit_code is None:
                    container_status = self.get_container_status(service_name)
                    service_report += f"  Exit Code: None (Container Status: {container_status})\n"

    def get_container_status(self, service_name):
    """
    Gets detailed status information for a specific container by service name.
    Used to provide better diagnostics for containers with None exit codes.

    :param service_name: The service name to check
    :return: Detailed status string or "unknown" if not found
    """
    try:
        # Try to get container ID
        result = subprocess.run(
            [
                "docker",
                "compose",
                "-f",
                str(self.rendered_services_network_config_file_path),
                "ps",
                "-q",
                service_name,
            ],
            capture_output=True,
            text=True,
        )

        if result.returncode != 0 or not result.stdout.strip():
            return "unknown"

        container_id = result.stdout.strip()

        # Get detailed status
        inspect_result = subprocess.run(
            [
                "docker",
                "inspect",
                "--format",
                "{{.State.Status}},{{.State.Running}},{{.State.ExitCode}},{{.State.Error}}",
                container_id,
            ],
            capture_output=True,
            text=True,
        )

        if inspect_result.returncode != 0:
            return "unknown"

        status_parts = inspect_result.stdout.strip().split(",")
        if len(status_parts) < 3:
            return "unknown"

        status = status_parts[0]
        running = status_parts[1] == "true"
        exit_code = status_parts[2]
        error = status_parts[3] if len(status_parts) > 3 else ""

        # Check logs for compilation errors
        logs_result = subprocess.run(
            [
                "docker",
                "logs",
                "--tail", "50",
                container_id,
            ],
            capture_output=True,
            text=True,
        )

        # Look for common error patterns in logs
        logs = logs_result.stdout + logs_result.stderr

        # Check for error log files first for most accurate detection
        status_file_path = os.path.join(self.output_dir, "logs", f"{service_name}_status.log")
        error_log_path = os.path.join(self.output_dir, "logs", f"{service_name}_error.log")
        compile_error_log = os.path.join(self.output_dir, "logs", f"{service_name}_compile_error.log")
        run_error_log = os.path.join(self.output_dir, "logs", f"{service_name}_run_error.log")

        # Track error information
        has_error = False
        error_type = None
        error_message = ""
        failed_command = None
        command_type = None

        # Check status file for detailed error information
        if os.path.exists(status_file_path):
            try:
                with open(status_file_path, "r") as f:
                    status_content = f.read()
                    if "ERROR_TYPE=" in status_content:
                        error_type = status_content.split("ERROR_TYPE=")[1].split("\n")[0].strip()
                        has_error = True
                    if "FAILED_COMMAND=" in status_content:
                        failed_cmd_parts = status_content.split("FAILED_COMMAND=\"")[1].split("\"")
                        if len(failed_cmd_parts) > 1:
                            failed_command = failed_cmd_parts[0]
                    if "COMMAND_TYPE=" in status_content:
                        command_type = status_content.split("COMMAND_TYPE=")[1].split("\n")[0].strip()
            except Exception as e:
                self.logger.error(f"Error reading status file: {str(e)}")

        # Check specific error logs
        if os.path.exists(error_log_path):
            try:
                with open(error_log_path, "r") as f:
                    error_content = f.read()
                    if error_content.strip():
                        has_error = True
                        error_message = error_content[:500] + "..." if len(error_content) > 500 else error_content
            except Exception as e:
                self.logger.error(f"Error reading error log: {str(e)}")

        # Enhanced pattern matching for various errors
        # TODO: Make more maintainable by using regex or a more structured approach
        error_patterns = {
            "compilation": ["error:", "compilation failed", "ivyc", "failed with code", 
                         "ivy compilation failed", "compiler error", "syntax error"],
            "runtime": ["segmentation fault", "core dumped", "abort", "killed", "terminated"],
            "system": ["no space left on device", "cannot allocate memory", "resource unavailable"],
            "timeout": ["timed out", "timeout", "time limit exceeded"],
            "network": ["connection refused", "network unreachable", "no route to host"]
        }

        detected_error_types = []
        for error_category, patterns in error_patterns.items():
            for pattern in patterns:
                if pattern in logs.lower():
                    has_error = True
                    detected_error_types.append(error_category)
                    # Extract context around the error
                    lines = logs.split('\n')
                    for i, line in enumerate(lines):
                        if pattern in line.lower():
                            start_idx = max(0, i - 5)
                            end_idx = min(len(lines), i + 5)
                            context = '\n'.join(lines[start_idx:end_idx])
                            if not error_message:  # Only set if not already set from error log
                                error_message = f"Found error pattern '{pattern}' in logs:\n{context}\n"
                            break
                    break

        # Check for status file indicators of compilation failure
        status_file_path = os.path.join(self.output_dir, "logs", f"{service_name}_status.log")
        compile_error_from_status = False
        if os.path.exists(status_file_path):
            try:
                with open(status_file_path, "r") as status_file:
                    status_content = status_file.read()
                    if "ERROR_TYPE=COMPILATION_FAILURE" in status_content:
                        compile_error_from_status = True
            except Exception as e:
                self.logger.error(f"Failed to read status file: {str(e)}")

        # Create detailed status string
        if running:
            # Check logs for compilation errors even if container is still running
            comp_file_path = os.path.join(self.output_dir, "logs", f"{service_name}_compilation.log")
            comp_error_path = os.path.join(self.output_dir, "logs", f"{service_name}_compile_error.log")

            if os.path.exists(comp_error_path):
                # Compilation explicitly failed
                try:
                    with open(comp_error_path, "r") as error_file:
                        error_content = error_file.read()
                        error_log_path = os.path.join(self.output_dir, "logs", f"{service_name}_detailed_errors.log")
                        with open(error_log_path, "w") as detailed_file:
                            detailed_file.write(error_content)
                            detailed_file.write("\n\n---- Full Container Logs ----\n\n")
                            detailed_file.write(logs)
                    return f"failed (compilation error detected, container still running but will fail, see {service_name}_detailed_errors.log)"
                except Exception as e:
                    self.logger.error(f"Failed to process compilation error logs: {str(e)}")

            return f"running (container is active with no exit code)"
        elif status == "exited" and exit_code == "0":
            # Even with exit code 0, check if we detected compilation errors in the logs
            has_compile_error = any("compilation" in err_type for err_type in detected_error_types)
            ivy_error_detected = "ivyc" in logs.lower() and "error:" in logs.lower()
            ivy_error_message = ""
            if ivy_error_detected:
                # Extract Ivy error message
                for line in logs.split("\n"):
                    if "ivyc" in line.lower() and "error:" in line.lower():
                        ivy_error_message = line
                        break

            if has_compile_error or ivy_error_detected or compile_error_from_status:
                error_log_path = os.path.join(self.output_dir, "logs", f"{service_name}_false_success.log")
                try:
                    with open(error_log_path, "w") as error_file:
                        error_file.write("Container exited with code 0 but compilation errors were detected\n\n")
                        if ivy_error_message:
                            error_file.write(ivy_error_message + "\n\n")
                        error_file.write("---- Full Container Logs ----\n\n")
                        error_file.write(logs)
                except Exception as e:
                    self.logger.error(f"Failed to write false success error log: {str(e)}")

                # Update the status file to reflect the actual failure
                try:
                    with open(status_file_path, "a") as status_file:
                        status_file.write("ERROR_TYPE=COMPILATION_FAILURE_MASKED\n")
                        status_file.write("FAILED=true\n")
                        status_file.write("EXIT_CODE=1\n")  # Override the exit code
                except Exception as e:
                    self.logger.error(f"Failed to update status file: {str(e)}")

                return f"failed (compilation errors detected despite exit code 0, see {service_name}_false_success.log)"
            return f"exited normally"
        elif status == "exited":
            # Check for compilation errors
            has_compile_error = any("compilation" in err_type for err_type in detected_error_types)
            ivy_error_detected = "ivyc" in logs.lower() and "error:" in logs.lower()
            ivy_error_message = ""
            if ivy_error_detected:
                # Extract Ivy error message
                for line in logs.split("\n"):
                    if "ivyc" in line.lower() and "error:" in line.lower():
                        ivy_error_message = line
                        break

            if has_compile_error or ivy_error_detected or compile_error_from_status:
                # Save the compilation error to a log file for review with enhanced context
                error_log_path = os.path.join(self.output_dir, "logs", f"{service_name}_compile_errors.log")
                try:
                    with open(error_log_path, "w") as error_file:
                        error_file.write(f"Container exited with code {exit_code}\n\n")
                        if ivy_error_message:
                            error_file.write(ivy_error_message + "\n\n")
                        error_file.write("---- Full Container Logs ----\n\n")
                        error_file.write(logs)
                except Exception as e:
                    self.logger.error(f"Failed to write compilation errors to log: {str(e)}")

                return f"failed (compilation error detected, see {service_name}_compile_errors.log)"
            elif error:
                return f"failed ({status}, error: {error})"
            else:
                return f"failed ({status}, exit code: {exit_code})"
        else:
            return f"{status}"

    except Exception as e:
        self.logger.error(f"Error getting container status for {service_name}: {str(e)}")
        return f"unknown (error: {str(e)})"

    def resolve_environment_variables(self, env_vars):
    """
    Resolves environment variables incrementally, ensuring no duplication
    and preserving unresolved tokens. Processes variables in dependency order.

    :param env_vars: dict, environment variables with potential references.
    :return: dict, resolved environment variables.
    """
    resolved_env = {}

    self.logger.debug("Initial environment variables:")
    for k, v in env_vars.items():
        self.logger.debug(f"{k}: {v}")

    for key, value in env_vars.items():
        if isinstance(value, str):
            resolved_value = value
            self.logger.debug(
                f"Resolving variable: {key} - Original value: {value}"
            )
            for (
                var_name,
                var_value,
            ) in resolved_env.items():  # Use already resolved variables
                if (
                    f"${{{var_name}}}" in resolved_value
                    or f"${var_name}" in resolved_value
                ):
                    resolved_value = resolved_value.replace(
                        f"${{{var_name}}}", var_value
                    )
                    resolved_value = resolved_value.replace(
                        f"${var_name}", var_value
                    )

    def teardown_environment(self):
    """
    Tears down the Docker Compose environment by bringing down services.
    First checks container exit codes to ensure they are captured before containers are removed.
    """
    # First check container exit codes and update service status
    self.check_container_exit_codes()

    # Generate a final test report summary
    self.generate_test_report_summary()

    # TODO: add a way to retrieve the logs, results, binary
    self.logger.info("Tearing down Docker Compose environment")
    with open(
        os.path.join(self.output_dir, "logs", "docker-compose-teardown.log"), "w"
    ) as log_file:
        with open(
            os.path.join(
                self.output_dir, "logs", "docker-compose-teardown.err.log"
            ),
            "w",
        ) as log_file_err:
            try:
                # For other network drivers, use docker-compose
                result = subprocess.run(
                    [
                        "docker",
                        "compose",
                        "-f",
                        self.services_network_config_file_path,
                        "down",
                    ],
                    capture_output=True,
                    text=True,  # Ensures that output is in string format
                )
                # Always write both stdout and stderr to the log file
                log_file.write(result.stdout)
                log_file_err.write(result.stderr)

                # Then check the return code
                if result.returncode != 0:
                    error_msg = f"Docker Compose down failed with code {result.returncode}: {result.stderr}"
                    self.logger.error(error_msg)
                    raise subprocess.CalledProcessError(
                        result.returncode, result.args, result.stdout, result.stderr
                    )

                os.system(
                    "docker volume prune -a -f"
                )  # TODO add that to docker manager
                self.logger.info(
                    "Docker Compose environment torn down successfully"
                )
            except Exception as e:
                self.logger.error(
                    f"Failed to tear down Docker Compose environment: {str(e)}"
                )
                log_file_err.write(
                    f"Exception during docker-compose down: {str(e)}\n"
                )
                raise
